{{/* share-buttons.html — inline share row for articles (news posts, press releases). Ported from the Cornwall Political Watch SharePanel/ShareButton React components: a "Share" button (Web Share API → clipboard fallback) followed by platform deep-links (X, Bluesky, Facebook, LinkedIn, WhatsApp, Email). Hugo gives us absolute URLs via .Permalink, so — unlike the React original — there is no client-side absolute-URL plumbing. Params (dict): url absolute page URL (pass .Permalink) title share title (pass .Title) text share body — used by WhatsApp/SMS/Email (pass excerpt/description) type analytics content_type, e.g. "news" or "press" (optional) Brand-icon SVGs are inline because Material Symbols has no platform logos. Place at most ONCE per page — the enhancement script uses event delegation but the markup is not meant to be duplicated. */}} {{- $url := .url -}} {{- $title := .title -}} {{- $text := .text | default .title -}} {{- $type := .type | default "page" -}} {{- $subject := chomp (printf "%v" $title) -}} {{- $shareText := chomp (printf "%v" $text) -}} {{- $x := printf "https://twitter.com/intent/tweet?%s" (querify "url" $url "text" $shareText) -}} {{- $bsky := printf "https://bsky.app/intent/compose?%s" (querify "text" (printf "%s %s" $shareText $url)) -}} {{- $fb := printf "https://www.facebook.com/sharer/sharer.php?%s" (querify "u" $url) -}} {{- $li := printf "https://www.linkedin.com/sharing/share-offsite/?%s" (querify "url" $url) -}} {{- $wa := printf "https://wa.me/?%s" (querify "text" (printf "%s %s" $shareText $url)) -}} {{- $email := printf "mailto:?%s" (querify "subject" $subject "body" (printf "%s\n\n%s" $shareText $url)) -}}
{{- $links := slice (dict "key" "x" "href" $x "label" "Share on X") (dict "key" "bluesky" "href" $bsky "label" "Share on Bluesky") (dict "key" "facebook" "href" $fb "label" "Share on Facebook") (dict "key" "linkedin" "href" $li "label" "Share on LinkedIn") (dict "key" "whatsapp" "href" $wa "label" "Share on WhatsApp") (dict "key" "email" "href" $email "label" "Share by email") -}} {{- range $links }} {{- if eq .key "x" -}} {{- else if eq .key "bluesky" -}} {{- else if eq .key "facebook" -}} {{- else if eq .key "linkedin" -}} {{- else if eq .key "whatsapp" -}} {{- else if eq .key "email" -}} {{- end -}} {{- end }}
